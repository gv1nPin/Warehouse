BEGIN;

-- =========================================================
-- 1. Справочники
-- =========================================================

CREATE TABLE IF NOT EXISTS public."Measurements"
(
    id               integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    measurement_name text NOT NULL UNIQUE CHECK (btrim(measurement_name) <> '')
);

CREATE TABLE IF NOT EXISTS public."Statuses"
(
    id          integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    status_name text NOT NULL UNIQUE CHECK (btrim(status_name) <> '')
);

CREATE TABLE IF NOT EXISTS public."Permissions"
(
    id              integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    permission_name text NOT NULL UNIQUE CHECK (btrim(permission_name) <> ''),
    description     text
);

CREATE TABLE IF NOT EXISTS public."Roles"
(
    id        integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    role_name text NOT NULL UNIQUE CHECK (btrim(role_name) <> '')
);

CREATE TABLE IF NOT EXISTS public."RolePermissions"
(
    role_id       integer NOT NULL REFERENCES public."Roles" (id) ON DELETE CASCADE,
    permission_id integer NOT NULL REFERENCES public."Permissions" (id) ON DELETE CASCADE,
    PRIMARY KEY (role_id, permission_id)
);

-- =========================================================
-- 2. Склады, сотрудники, товары, остатки
-- =========================================================

CREATE TABLE IF NOT EXISTS public."Warehouses"
(
    id         integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title      text    NOT NULL CHECK (btrim(title) <> ''),
    address    text    NOT NULL CHECK (btrim(address) <> ''),
    is_deleted boolean NOT NULL DEFAULT false
);

CREATE TABLE IF NOT EXISTS public."Employees"
(
    id            integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    first_name    text    NOT NULL CHECK (btrim(first_name) <> ''),
    last_name     text    NOT NULL CHECK (btrim(last_name) <> ''),
    warehouse_id  integer NOT NULL REFERENCES public."Warehouses" (id),
    role_id       integer NOT NULL REFERENCES public."Roles" (id),
    is_deleted    boolean NOT NULL DEFAULT false,
    login         text    NOT NULL UNIQUE CHECK (btrim(login) <> ''),
    password_hash text    NOT NULL
);

CREATE TABLE IF NOT EXISTS public."Products"
(
    id             integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    article_number text    NOT NULL UNIQUE CHECK (btrim(article_number) <> ''),
    product_name   text    NOT NULL CHECK (btrim(product_name) <> ''),
    measurement_id integer NOT NULL REFERENCES public."Measurements" (id),
    is_deleted     boolean NOT NULL DEFAULT false
);

CREATE TABLE IF NOT EXISTS public."StockOnWarehouse"
(
    warehouse_id      integer        NOT NULL REFERENCES public."Warehouses" (id),
    product_id        integer        NOT NULL REFERENCES public."Products" (id),
    quantity          numeric(12, 3) NOT NULL DEFAULT 0 CHECK (quantity >= 0),
    reserved_quantity numeric(12, 3) NOT NULL DEFAULT 0 CHECK (reserved_quantity >= 0),
    last_updated      timestamptz    NOT NULL DEFAULT now(),
    PRIMARY KEY (warehouse_id, product_id),
    CONSTRAINT "ck_StockOnWarehouse_reserved_le_quantity" CHECK (reserved_quantity <= quantity)
);

-- =========================================================
-- 3. Перевозки
-- =========================================================

CREATE TABLE IF NOT EXISTS public."Shipments"
(
    id           integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    status_id    integer     NOT NULL REFERENCES public."Statuses" (id),
    planned_date date        NOT NULL,
    creator_id   integer     NOT NULL REFERENCES public."Employees" (id),
    created_at   timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public."ShipmentStages"
(
    id                integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    shipment_id       integer NOT NULL REFERENCES public."Shipments" (id) ON DELETE CASCADE,
    stage_order       integer NOT NULL CHECK (stage_order > 0),
    status_id         integer NOT NULL REFERENCES public."Statuses" (id),
    from_warehouse_id integer NOT NULL REFERENCES public."Warehouses" (id),
    to_warehouse_id   integer NOT NULL REFERENCES public."Warehouses" (id),
    acceptor_id       integer REFERENCES public."Employees" (id),
    driver_id         integer REFERENCES public."Employees" (id),
    sent_at           timestamptz,
    received_at       timestamptz,
    CONSTRAINT "uq_ShipmentStages_shipment_order" UNIQUE (shipment_id, stage_order),
    CONSTRAINT "ck_ShipmentStages_different_warehouses" CHECK (from_warehouse_id <> to_warehouse_id),
    CONSTRAINT "ck_ShipmentStages_received_after_sent"
        CHECK (received_at IS NULL OR (sent_at IS NOT NULL AND received_at >= sent_at))
);

CREATE TABLE IF NOT EXISTS public."StageItems"
(
    id                integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    stage_id          integer        NOT NULL REFERENCES public."ShipmentStages" (id) ON DELETE CASCADE,
    product_id        integer        NOT NULL REFERENCES public."Products" (id),
    document_quantity numeric(12, 3) NOT NULL CHECK (document_quantity > 0),
    actual_quantity   numeric(12, 3) CHECK (actual_quantity >= 0),
    comment           text,
    CONSTRAINT "uq_StageItems_stage_product" UNIQUE (stage_id, product_id)
);

-- Документы этапа (накладные, доверенности, фото). Сам файл лежит в хранилище
-- web-слоя, здесь только его данные. Без документа черновик не зарезервировать.
CREATE TABLE IF NOT EXISTS public."StageDocuments"
(
    id           integer     GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    stage_id     integer     NOT NULL REFERENCES public."ShipmentStages" (id) ON DELETE CASCADE,
    file_name    text        NOT NULL CHECK (btrim(file_name) <> ''),
    storage_path text        NOT NULL CHECK (btrim(storage_path) <> ''),
    content_type text,
    size_bytes   bigint      CHECK (size_bytes >= 0),
    uploaded_by  integer     NOT NULL REFERENCES public."Employees" (id),
    uploaded_at  timestamptz NOT NULL DEFAULT now()
);

-- =========================================================
-- 4. Индексы
-- =========================================================

CREATE INDEX IF NOT EXISTS "ix_Employees_role_id"                ON public."Employees" (role_id);
CREATE INDEX IF NOT EXISTS "ix_Employees_warehouse_id"           ON public."Employees" (warehouse_id);
CREATE INDEX IF NOT EXISTS "ix_Products_measurement_id"          ON public."Products" (measurement_id);
CREATE INDEX IF NOT EXISTS "ix_RolePermissions_permission_id"    ON public."RolePermissions" (permission_id);
CREATE INDEX IF NOT EXISTS "ix_StockOnWarehouse_product_id"      ON public."StockOnWarehouse" (product_id);
CREATE INDEX IF NOT EXISTS "ix_Shipments_creator_id"             ON public."Shipments" (creator_id);
CREATE INDEX IF NOT EXISTS "ix_Shipments_status_id"              ON public."Shipments" (status_id);
CREATE INDEX IF NOT EXISTS "ix_ShipmentStages_status_id"         ON public."ShipmentStages" (status_id);
CREATE INDEX IF NOT EXISTS "ix_ShipmentStages_from_warehouse_id" ON public."ShipmentStages" (from_warehouse_id);
CREATE INDEX IF NOT EXISTS "ix_ShipmentStages_to_warehouse_id"   ON public."ShipmentStages" (to_warehouse_id);
CREATE INDEX IF NOT EXISTS "ix_ShipmentStages_acceptor_id"       ON public."ShipmentStages" (acceptor_id);
CREATE INDEX IF NOT EXISTS "ix_ShipmentStages_driver_id"         ON public."ShipmentStages" (driver_id);
CREATE INDEX IF NOT EXISTS "ix_StageDocuments_stage_id"          ON public."StageDocuments" (stage_id);
CREATE INDEX IF NOT EXISTS "ix_StageDocuments_uploaded_by"       ON public."StageDocuments" (uploaded_by);
CREATE INDEX IF NOT EXISTS "ix_StageItems_product_id"            ON public."StageItems" (product_id);
